2-Way Google Sheets Auto-Sync Script (`Updated DTR`)

1. Open your Updated DTR Google Sheet.
2. Click Extensions → Apps Script and paste the code below.
3. Click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
4. Paste the Web App URL into the Sheet Sync tab! 

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Update Individual Crew Tab (Maricel, Pamela, or Aileen)
  const crewSheet = ss.getSheetByName(data.crew);
  if (crewSheet) {
    const values = crewSheet.getDataRange().getValues();
    let updated = false;
    for (let i = 1; i < values.length; i++) {
      const rowDate = Utilities.formatDate(new Date(values[i][0]), "GMT+8", "yyyy-MM-dd");
      if (rowDate === data.date || String(values[i][0]).indexOf(data.date) !== -1) {
        crewSheet.getRange(i + 1, 3).setValue(data.timeIn);
        crewSheet.getRange(i + 1, 4).setValue(data.timeOut);
        crewSheet.getRange(i + 1, 5).setValue(data.status);
        updated = true;
        break;
      }
    }
    if (!updated) {
      crewSheet.appendRow([data.date, data.day, data.timeIn, data.timeOut, data.status]);
    }
  }

  // 2. Update Master "Daily Time Record" Sheet
  const masterSheet = ss.getSheetByName("Daily Time Record");
  if (masterSheet) {
    const colMap = { "Maricel": 3, "Pamela": 5, "Aileen": 7 };
    const startCol = colMap[data.crew];
    if (startCol) {
      const mValues = masterSheet.getDataRange().getValues();
      for (let r = 5; r < mValues.length; r++) {
        const cellText = String(mValues[r][0] || mValues[r][1]);
        if (cellText.indexOf(data.date) !== -1) {
          masterSheet.getRange(r + 1, startCol).setValue(data.timeIn);
          masterSheet.getRange(r + 1, startCol + 1).setValue(data.timeOut);
          break;
        }
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const result = {};
  ["Maricel", "Pamela", "Aileen"].forEach(function(crew) {
    const sheet = ss.getSheetByName(crew);
    if (sheet) result[crew] = sheet.getDataRange().getDisplayValues();
  });
  return ContentService.createTextOutput(JSON.stringify({ ok: true, data: result }))
    .setMimeType(ContentService.MimeType.JSON);
}

How to Activate & Publish via GitHub Pages

Step 0: Download Your Ready-to-Upload `index.html`

If you pasted your Google Sheet Web App URL in the Sheet Sync tab, it will be pre-saved inside this downloaded file automatically!

Step 1: Create a New Repository on GitHub

1. Sign in to github.com (or create a free account).
2. Tap the + icon in the top-right corner and select New repository.
3. Name your repository crew-dtr.
4. Select Public, check Add a README file, and click Create repository.

Step 2: Upload the `index.html` File

1. Inside your new crew-dtr repository, click Add file → Upload files.
2. Drag and drop your downloaded index.html file (make sure the filename is lowercase index.html).
3. Scroll down and click the green Commit changes button.

Step 3: Turn On GitHub Pages

1. Click the Settings tab at the top of your repository.
2. In the left sidebar, click Pages.
3. Under Build and deployment → Branch, change None to main (and keep folder as /(root)), then click Save.
4. Wait about 60 seconds and refresh the page—your live link will appear at the top: https://YOUR-USERNAME.github.io/crew-dtr/

Step 4: Install on Crew Phones (Online & Offline Ready)

• Send your github.io link to Maricel, Pamela, and Aileen.
• On Android (Chrome): Tap Menu (⋮) → Add to Home screen.
• On iPhone (Safari): Tap Share → Add to Home Screen.
• Crew members can now tap SIGN IN and SIGN OUT daily even when offline, and it will auto-sync to your Updated DTR Google Sheet!
